//Decleare global or statis instance
static Settings_t g_settings;

//Load settings at boot
void system_init(void)
{
    settings_load(&g_settings);

    /* Optional: validate checksum */
    if (g_settings.checksum != settings_checksum(&g_settings)) {
        // Flash data corrupted → restore defaults
        g_settings.magic = 0xDEADBEEF;
        g_settings.baudrate = 115200;
        g_settings.device_id = 1;
        g_settings.flags = 0;
        g_settings.checksum = settings_checksum(&g_settings);
    }
}


//Use settings anywhere in code 
void uart_init_from_settings(void)
{
    uart_set_baudrate(g_settings.baudrate);
}

uint8_t get_device_id(void)
{
    return g_settings.device_id;
}


//Modify settings at runtime 
void cmd_set_baudrate(uint32_t new_baud)
{
    if (g_settings.baudrate != new_baud) {
        g_settings.baudrate = new_baud;
        g_settings.checksum = settings_checksum(&g_settings);
        settings_save(&g_settings);
    }
}



//Save settings when exiting command mode
void config_mode_exit(void)
{
    g_settings.checksum = settings_checksum(&g_settings);
    settings_save(&g_settings);
}




//Full minimal code example of main.c
#include "Flash.h"

static Settings_t g_settings;

int main(void)
{
    system_clock_init();
    gpio_init();
    uart_init();

    settings_load(&g_settings);

    if (g_settings.checksum != settings_checksum(&g_settings)) {
        g_settings.magic = 0xDEADBEEF;
        g_settings.baudrate = 115200;
        g_settings.device_id = 1;
        g_settings.flags = 0;
    }

    uart_set_baudrate(g_settings.baudrate);

    while (1) {
        if (user_pressed_button()) {
            g_settings.device_id++;
            g_settings.checksum = settings_checksum(&g_settings);
            settings_save(&g_settings);
        }
    }
}




RULES

1. Always load once at boot
    settings_load(&settings);

2. Always update checksum before saving
    settings.checksum = settings_checksum(&settings);

3.  Always modify the ram copy never write directly to flash
